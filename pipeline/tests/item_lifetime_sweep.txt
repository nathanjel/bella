class I:
    """An iterator that says when its __next__ is called; each item says when it goes. What a consumer written in C lets go
    of before it asks for the next item is gone before the next "next"."""
    def __init__(self, kind, count): self.kind, self.count, self.i = kind, count, 0
    def __iter__(self): return self
    def __next__(self):
        self.i += 1
        print("next", self.i)
        if self.i > self.count: raise StopIteration
        return MAKE[self.kind](self.i)
    def __del__(self): print("drop iterator")
