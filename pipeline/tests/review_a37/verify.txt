if sys.argv[1]=='release':
    import itertools
    class X:
        def __del__(self): print('drop')
    def source():
        yield 1
        yield X()
        raise KeyboardInterrupt()
    it=source()
    w=itertools.pairwise(it)
    try:
        for x in w: del x
    except KeyboardInterrupt: print('caught')
    print('release')
    del w,it
    print('end')
