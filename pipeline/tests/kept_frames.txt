import gc
import traceback
class T:
    def __init__(s, n): s.n = n
    def __del__(s): print("  released", s.n)
def inner():
    i = T("inner local")
    try: raise ValueError("x")
    except ValueError as e: return e
def outer():
    t = T("outer local")
    return inner()
print("a caller of a frame that kept its exception:")
kept = outer()
print("  outer returned")
del kept
print("  gone")
def raises():
    r = T("raiser local")
    raise KeyError("k")
def middle():
    m = T("middle local")
    raises()
def top():
    t = T("top local")
    try: middle()
    except KeyError as e: return e
print("an exception that passed two frames, caught by a third:")
kept = top()
print("  top returned")
del kept
print("  gone")
def gen(out):
    g = T("generator local")
    try: raise ValueError("y")
    except ValueError as e: out.append(e)
    yield 1
    print("  after the yield:", g.n)
    yield 2
out = []
it = gen(out)
print("a generator that kept its exception from its first resume:")
print(" ", next(it)); print(" ", next(it)); print(" ", next(it, "end"))
del it
print("  the generator is gone")
out.clear()
print("  the list is empty")
it = gen(out)
next(it)
print("closed while suspended:")
del it
print("  the generator is gone")
out.clear()
print("  the list is empty")

    try:
        raise ValueError
    except ValueError:
        yield 1
        raise
it = g2(); next(it)
try: next(it)
except ValueError as e:
    tb = e.__traceback__; n = 0
    while tb: n += 1; tb = tb.tb_next
    print("entries after a bare raise past a yield:", n)

def two(out):
    t = T("a local of a frame with two kept exceptions")
    try: raise ValueError("a")
    except ValueError as e: out.append(e)
    try: raise KeyError("b")
    except KeyError as e: out.append(e)
for first in (0, 1):
    out = []
    two(out)
    print("two kept, the", "older" if first == 0 else "newer", "let go of first:")
    del out[first]

    print("  both are gone")

tb = returns_tb()
