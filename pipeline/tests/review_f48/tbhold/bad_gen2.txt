"""Raises after it gave away a suspended generator and deleted the generator's function: the generator alone holds the namespace."""
import sys
from collections.abc import Iterator

from watch import Marker, kept, watched

held = Marker("gen2.held")
watched(sys.modules[__name__], "bad_gen2")


def gen(d: Marker = Marker("gen2.default")) -> Iterator[str]:
    local = Marker("gen2.local")
    yield held.name + local.name + d.name
    yield "second"


g = gen()
next(g)
kept.append(g)
del g, gen
raise ValueError("bad_gen2")
