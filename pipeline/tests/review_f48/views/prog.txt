Maker = Callable[[bytes], memoryview]


def contiguous(data: bytes) -> memoryview:
    return memoryview(data)


def writable(data: bytes) -> memoryview:
    return memoryview(bytearray(data))


def strided(data: bytes) -> memoryview:
    wide = bytearray(len(data) * 2)
    wide[::2] = data
    return memoryview(wide)[::2]


KINDS: dict[str, Maker] = {"c": contiguous, "w": writable, "s": strided}


def scratch() -> str:
    return "q40-p2-" + str(os.getpid()) + ".bin"


def with_fd(data: bytes, use: Callable[[int], object]) -> object:
    name = scratch()
    with open(name, "wb") as out:
        out.write(data)
    fd = os.open(name, os.O_RDWR)
    try:
        got = use(fd)
        with open(name, "rb") as back:
            return got, back.read()
    finally:
        os.close(fd)
        os.unlink(name)


def into(view: memoryview, use: Callable[[memoryview], object]) -> object:
    got = use(view)
    return got, view.tobytes()


def xattr(view: memoryview) -> object:
    name = scratch()
    with open(name, "wb") as out:
        out.write(b"x")
    try:
        os.setxattr(name, "user.q40", view)
        return os.getxattr(name, "user.q40")
    finally:
        os.unlink(name)


def sha_update(view: memoryview) -> str:
    h = hashlib.sha256()
    h.update(view)
    return h.hexdigest()[:8]


def sha3_update(view: memoryview) -> str:
    h = hashlib.sha3_256()
    h.update(view)
    return h.hexdigest()[:8]


def hmac_update(view: memoryview) -> str:
    h = hmac.new(b"k", None, "sha256")
    h.update(view)
    return h.hexdigest()[:8]


def compress_object(view: memoryview) -> int:
    c = zlib.compressobj()
    return len(c.compress(view) + c.flush())


def decompress_object(view: memoryview) -> bytes:
    return zlib.decompressobj().decompress(view)


def writelines(view: memoryview) -> bytes:
    out = io.BytesIO()
    out.writelines([view, view])
    return out.getvalue()


def buffered_write(view: memoryview) -> bytes:
    sink = io.BytesIO()
    writer = io.BufferedWriter(sink, 4)  # type: ignore[arg-type]
    writer.write(view)
    writer.flush()
    return sink.getvalue()


def file_write(view: memoryview) -> object:
    name = scratch()
    try:
        with open(name, "wb") as out:
            n = out.write(view)
        with open(name, "rb") as back:
            return n, back.read()
    finally:
        os.unlink(name)


def fileio_write(view: memoryview) -> object:
    name = scratch()
    try:
        with io.FileIO(name, "w") as out:
            n = out.write(view)
        with open(name, "rb") as back:
            return n, back.read()
    finally:
        os.unlink(name)


def text_write(view: memoryview) -> object:
    return io.TextIOWrapper(io.BytesIO()).write(view)  # type: ignore[arg-type]


def incremental(view: memoryview) -> str:
    return codecs.getincrementaldecoder("utf-8")().decode(view)


def path_write(view: memoryview) -> object:
    import pathlib
    name = scratch()
    try:
        n = pathlib.Path(name).write_bytes(view)
        return n, pathlib.Path(name).read_bytes()
    finally:
        os.unlink(name)
