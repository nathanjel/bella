def main() -> None:
    name = sys.argv[1]
    if name == "__main__":
        watched(sys.modules["__main__"], name)
    else:
        watched(importlib.import_module(name), name)
    sys.__dict__["q44refs"] = watch.refs
    LOUD.append(True)
    print("the end", flush=True)


main()
