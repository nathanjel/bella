def tb_sanity() -> None:
    for label, name, package in (("empty", "", None), ("beyond", "..x", "pkgf"), ("no package", ".x", None), ("relative missing", ".nothere", "pkgf")):
        try:
            importlib.import_module(name, package)
        except Exception as e:
            entries(label, e)
            print(type(e).__name__, e)
    for label, level in (("negative level", -1), ("level without globals", 1)):
        try:
            importlib.__import__("x", None, None, (), level)
        except Exception as e:
            entries(label, e)
            print(type(e).__name__, e)
    try:
        importlib.__import__(5)  # type: ignore[arg-type]
    except Exception as e:
        entries("a name that is no str", e)
        print(type(e).__name__, e)

def down(n: int, what: str) -> int:
    try:
        if what == "repr":
            repr(watch)
        elif what == "cached":
            import watch as again  # noqa: F401
        elif what == "fromlist":
            from pkgf import okname  # noqa: F401
        elif what == "byname":
            importlib.import_module("watch")
        elif what == "dunder":
            __import__("pkgf.okname2")
    except RecursionError:
        return n
    except ImportError:
        pass
    try:
        return down(n + 1, what)
    except RecursionError:
        return -n


def depth_calls() -> None:
    import pkgf  # noqa: F401
    limit = sys.getrecursionlimit()
    for what in ("none", "repr", "cached", "fromlist", "byname"):
        sys.setrecursionlimit(70)
        reached = down(0, what)
        sys.setrecursionlimit(limit)
        print("the level at which", what, "first met the limit", reached)

class ImportsByName(unittest.TestCase):

def unittest_rows() -> None:
    result = unittest.TestResult()
    suite = unittest.defaultTestLoader.loadTestsFromTestCase(ImportsByName)
    suite.run(result)
    for test, text in result.errors:
        print(test.id().rsplit(".", 1)[-1])
        for line in text.splitlines():
            if line.startswith("  File"):
                print("   ", line.strip().removeprefix('File "').rsplit("/", 1)[-1])   # the file's base name on both sides
            elif not line.startswith("    "):
                print("   ", line)
