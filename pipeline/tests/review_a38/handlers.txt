def note(s): print(s,type(sys.exception()).__name__)
def summary(e):
    tb=e.__traceback__
    n=0
    while tb:
        n+=1
        tb=tb.tb_next
    print(type(e).__name__,type(e.__context__).__name__,type(e.__cause__).__name__,n)

class F(io.IOBase):

        note('flush')
        self.depth+=1

            io.IOBase.__setattr__(self,k,v)
